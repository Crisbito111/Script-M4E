/**
 * Se ejecuta con un trigger diario (ej. todos los días a las 8pm).
 * Busca la fila de hoy en el Sheet, arma el mensaje del gasto de hoy,
 * y lo envía por correo.
 */
function enviarDatoSecuencial() {
  var cuerpo = construirMensajeGastoHoy();
  var asunto = 'OE WEBONAZO MIRA TUS GASTOS';
  MailApp.sendEmail(EMAIL_DESTINO, asunto, cuerpo);
}


/**
 * Green API llama a esta función automáticamente (POST) cada vez que
 * te llega un mensaje de texto al WhatsApp vinculado (en tu caso, cuando
 * te escribes a ti mismo). Responde con el gasto de hoy por el mismo chat.
 */
function doPost(e) {
  var payload = JSON.parse(e.postData.contents);

  // Green API notifica varios tipos de eventos (mensaje entregado, leído,
  // etc.). Solo nos interesa un mensaje de texto entrante; el resto se ignora.
  if (payload.typeWebhook !== 'incomingMessageReceived') {
    return ContentService.createTextOutput('ignorado');
  }

  var chatId = payload.senderData.chatId;
  var cuerpo = construirMensajeGastoHoy();

  enviarWhatsApp(chatId, cuerpo);

  return ContentService.createTextOutput('ok');
}


/**
 * Lógica compartida por ambos flujos (email y WhatsApp): busca el gasto
 * de hoy en el Sheet y arma el texto del mensaje. Centralizarla acá evita
 * tener la misma lógica duplicada en dos funciones distintas.
 */
function construirMensajeGastoHoy() {
  var gastoHoy = calcularGastoDeHoy();

  return (gastoHoy === 0)
    ? 'No registraste gastos hoy.'
    : 'ESPERO QUE HAYA VALIDO LA PENA ESTE DÍA, PORQUE GASTASTE: S/.' + gastoHoy.toFixed(2);
}


/**
 * Busca en el Sheet la fila cuya fecha es la de hoy y devuelve el gasto
 * de esa fila como número. Devuelve 0 si no hay fila para hoy todavía
 * o si la celda de gasto está vacía.
 */
function calcularGastoDeHoy() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();

  var colFecha = columnLetterToIndex(COL_LETRA_FECHA);
  var colGasto = columnLetterToIndex(COL_LETRA_GASTO);

  // Trae solo las columnas entre fecha y gasto, hasta la última fila con datos.
  // Como el Sheet crece indefinidamente, esto simplemente traerá más filas
  // con el tiempo, sin que haya que tocar el código.
  var lastRow = sheet.getLastRow();
  var datos = sheet.getRange(1, colFecha + 1, lastRow, (colGasto - colFecha) + 1).getValues();

  var hoy = new Date();
  hoy.setHours(0, 0, 0, 0);

  // i = 1 para saltar la fila de encabezados (fila 1 del Sheet).
  for (var i = 1; i < datos.length; i++) {
    var fechaFila = parsearFecha(datos[i][0]); // primera columna del rango = fecha
    if (!fechaFila) continue;

    if (fechaFila.getTime() === hoy.getTime()) {
      var gasto = datos[i][datos[i].length - 1]; // última columna del rango = gasto
      return gasto === '' ? 0 : Number(gasto);
    }
  }

  return 0; // no hay fila para hoy todavía
}


/**
 * Envía un mensaje de texto por WhatsApp usando la API REST de Green API.
 */
function enviarWhatsApp(chatId, mensaje) {
  var url = GREEN_API_URL + '/waInstance' + GREEN_API_ID_INSTANCE
    + '/sendMessage/' + GREEN_API_TOKEN;

  var opciones = {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({ chatId: chatId, message: mensaje })
  };

  UrlFetchApp.fetch(url, opciones);
}


/**
 * Convierte una letra de columna de Sheets (ej. 'H') a índice de arreglo
 * base 0 (ej. 7). A=0, B=1, ..., H=7, J=9.
 */
function columnLetterToIndex(letra) {
  return letra.toUpperCase().charCodeAt(0) - 'A'.charCodeAt(0);
}


/**
 * Convierte el valor crudo de una celda de fecha en un objeto Date
 * normalizado (sin horas), sin importar si Sheets lo entregó como:
 *   a) un objeto Date real (columna con formato de fecha), o
 *   b) un texto "dd/mm" o "dd/mm/yyyy" (columna con formato de texto).
 * Devuelve null si la celda está vacía o no se pudo interpretar.
 */
function parsearFecha(valorCelda) {
  if (!valorCelda) return null;

  var fecha;

  if (valorCelda instanceof Date) {
    fecha = new Date(valorCelda.getTime());
  } else {
    var texto = String(valorCelda).trim();
    var partes = texto.split('/');
    if (partes.length < 2) return null;

    var dia = parseInt(partes[0], 10);
    var mes = parseInt(partes[1], 10) - 1;
    var anio = partes[2] ? parseInt(partes[2], 10) : new Date().getFullYear();

    if (isNaN(dia) || isNaN(mes)) return null;
    fecha = new Date(anio, mes, dia);
  }

  fecha.setHours(0, 0, 0, 0);
  return fecha;
}
